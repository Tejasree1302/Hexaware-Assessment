USE ROLE ACCOUNTADMIN;
--create Data Base
CREATE OR REPLACE DATABASE RETAIL_DB;

--create schema
CREATE OR REPLACE SCHEMA RETAIL_DB.ORDERS_DEMO;

--create warehouse
CREATE OR REPLACE WAREHOUSE RETAIL_LOAD_WH
WITH
WAREHOUSE_SIZE = 'XSMALL'
AUTO_SUSPEND = 60
AUTO_RESUME = TRUE
INITIALLY_SUSPENDED = TRUE;

--create role
CREATE OR REPLACE ROLE RETAIL_DATA_ENGINEER_ROLE;

--create user
CREATE OR REPLACE USER RETAIL_ENGINEER_USER
PASSWORD = 'Retail@123'
DEFAULT_ROLE = RETAIL_DATA_ENGINEER_ROLE
DEFAULT_WAREHOUSE = RETAIL_LOAD_WH
MUST_CHANGE_PASSWORD = FALSE;


--Grant Role to User
GRANT ROLE RETAIL_DATA_ENGINEER_ROLE
TO USER RETAIL_ENGINEER_USER;


--Grant Permissions
GRANT USAGE ON WAREHOUSE RETAIL_LOAD_WH
TO ROLE RETAIL_DATA_ENGINEER_ROLE;

GRANT USAGE ON DATABASE RETAIL_DB
TO ROLE RETAIL_DATA_ENGINEER_ROLE;

GRANT USAGE ON SCHEMA RETAIL_DB.ORDERS_DEMO
TO ROLE RETAIL_DATA_ENGINEER_ROLE;

GRANT CREATE TABLE ON SCHEMA RETAIL_DB.ORDERS_DEMO
TO ROLE RETAIL_DATA_ENGINEER_ROLE;

GRANT CREATE STAGE ON SCHEMA RETAIL_DB.ORDERS_DEMO
TO ROLE RETAIL_DATA_ENGINEER_ROLE;

GRANT CREATE PIPE ON SCHEMA RETAIL_DB.ORDERS_DEMO
TO ROLE RETAIL_DATA_ENGINEER_ROLE;

GRANT CREATE TASK ON SCHEMA RETAIL_DB.ORDERS_DEMO
TO ROLE RETAIL_DATA_ENGINEER_ROLE;

GRANT CREATE STREAM ON SCHEMA RETAIL_DB.ORDERS_DEMO
TO ROLE RETAIL_DATA_ENGINEER_ROLE;

USE DATABASE RETAIL_DB;
USE SCHEMA ORDERS_DEMO;
USE WAREHOUSE RETAIL_LOAD_WH;

--create file format
CREATE OR REPLACE FILE FORMAT RETAIL_CSV_FORMAT
TYPE = CSV
SKIP_HEADER = 1
FIELD_OPTIONALLY_ENCLOSED_BY = '"';


CREATE OR REPLACE STAGE RETAIL_AZURE_STAGE
URL='azure://hexaretailstore.blob.core.windows.net/retail-data/raw/orders/'
CREDENTIALS=(
AZURE_SAS_TOKEN='sv=2026-02-06&ss=bfqt&srt=sco&sp=rwdlacupyx&se=2026-05-21T11:54:19Z&st=2026-05-21T09:55:19Z&spr=https&sig=iT5MwcwmsMdGozQ7ZGKeNfr3tnrQMoFmIb%2B2ctxWn0U%3D'
)
FILE_FORMAT = RETAIL_CSV_FORMAT;


LIST @RETAIL_AZURE_STAGE;

--Create Raw Table
CREATE OR REPLACE TABLE ORDERS_RAW
(
ORDER_ID NUMBER,
CUSTOMER_NAME STRING,
PRODUCT STRING,
AMOUNT NUMBER,
ORDER_DATE DATE
);


--Create Final Table
CREATE OR REPLACE TABLE ORDERS_FINAL
(
ORDER_ID NUMBER,
CUSTOMER_NAME STRING,
PRODUCT STRING,
AMOUNT NUMBER,
ORDER_DATE DATE
);

--Load Initial CSV into Raw Table
COPY INTO ORDERS_RAW
FROM @RETAIL_AZURE_STAGE
FILES = ('retail_orders_initial.csv');

TRUNCATE TABLE ORDERS_RAW;
LIST @RETAIL_AZURE_STAGE;


--Load Initial CSV into Raw Table
COPY INTO ORDERS_RAW
FROM @RETAIL_AZURE_STAGE
FILES=('retail_orders_initial.csv');


SELECT * FROM ORDERS_RAW;

--Insert into Final Table
INSERT INTO ORDERS_FINAL
SELECT * FROM ORDERS_RAW;


--Verify Final Data
SELECT * FROM ORDERS_FINAL;


--Create Snowpipe
CREATE OR REPLACE PIPE ORDERS_PIPE
AUTO_INGEST = FALSE
AS
COPY INTO ORDERS_RAW
FROM @RETAIL_AZURE_STAGE
FILE_FORMAT = RETAIL_CSV_FORMAT;


--create stream
CREATE OR REPLACE STREAM ORDERS_RAW_STREAM
ON TABLE ORDERS_RAW;

--Create Task
MERGE INTO ORDERS_FINAL T
USING
(
SELECT *
FROM ORDERS_RAW_STREAM
) S

ON T.ORDER_ID = S.ORDER_ID

WHEN NOT MATCHED THEN
INSERT
(
ORDER_ID,
CUSTOMER_NAME,
PRODUCT,
AMOUNT,
ORDER_DATE)
VALUES
(
S.ORDER_ID,
S.CUSTOMER_NAME,
S.PRODUCT,
S.AMOUNT,
S.ORDER_DATE);

SELECT * FROM ORDERS_FINAL;

ALTER PIPE ORDERS_PIPE REFRESH;


SELECT * FROM ORDERS_RAW;
