USE ROLE ACCOUNTADMIN;

--creating warehouses MARKETING_WH, FINANCE_WH, ETL_WH
CREATE OR REPLACE WAREHOUSE MARKETING_WH
WITH
WAREHOUSE_SIZE = 'SMALL'
AUTO_SUSPEND = 60
AUTO_RESUME = TRUE
INITIALLY_SUSPENDED = TRUE;


CREATE OR REPLACE WAREHOUSE FINANCE_WH
WITH
WAREHOUSE_SIZE = 'XSMALL'
AUTO_SUSPEND = 60
AUTO_RESUME = TRUE
INITIALLY_SUSPENDED = TRUE;


CREATE OR REPLACE WAREHOUSE ETL_WH
WITH
WAREHOUSE_SIZE = 'MEDIUM'
AUTO_SUSPEND = 60
AUTO_RESUME = TRUE
INITIALLY_SUSPENDED = TRUE;

SHOW WAREHOUSES;

--Create Roles
CREATE OR REPLACE ROLE MARKETING_ANALYST;


CREATE OR REPLACE ROLE FINANCE_USER;


CREATE OR REPLACE ROLE ETL_DEVELOPER;

--Marketing Role
GRANT USAGE
ON WAREHOUSE MARKETING_WH
TO ROLE MARKETING_ANALYST;

--Finance Role
GRANT USAGE
ON WAREHOUSE FINANCE_WH
TO ROLE FINANCE_USER;

--ETL Role
GRANT USAGE
ON WAREHOUSE ETL_WH
TO ROLE ETL_DEVELOPER;

--Create Marketing User
CREATE OR REPLACE USER MKT_USER1
PASSWORD = 'MktUser@123'
DEFAULT_ROLE = MARKETING_ANALYST
DEFAULT_WAREHOUSE = MARKETING_WH
MUST_CHANGE_PASSWORD = FALSE;

--Create Finance User
CREATE OR REPLACE USER FIN_USER1
PASSWORD = 'FinUser@123'
DEFAULT_ROLE = FINANCE_USER
DEFAULT_WAREHOUSE = FINANCE_WH
MUST_CHANGE_PASSWORD = FALSE;

--Create ETL User
CREATE OR REPLACE USER ETL_USER1
PASSWORD = 'EtlUser@123'
DEFAULT_ROLE = ETL_DEVELOPER
DEFAULT_WAREHOUSE = ETL_WH
MUST_CHANGE_PASSWORD = FALSE;

GRANT ROLE MARKETING_ANALYST TO USER MKT_USER1;

GRANT ROLE FINANCE_USER TO USER FIN_USER1;

GRANT ROLE ETL_DEVELOPER TO USER ETL_USER1;

SHOW USERS;

--create DATABASE
CREATE OR REPLACE DATABASE RETAIL_MARKETING_DB;

USE DATABASE RETAIL_MARKETING_DB;

USE SCHEMA PUBLIC;


ALTER WAREHOUSE ETL_WH SUSPEND;

USE WAREHOUSE FINANCE_WH;

--Create Table
CREATE OR REPLACE TABLE CAMPAIGN_SALES AS
SELECT
    SS.SS_SOLD_DATE_SK,
    D.D_DATE AS SALE_DATE,
    SS.SS_CUSTOMER_SK,
    C.C_FIRST_NAME,
    C.C_LAST_NAME,
    SS.SS_ITEM_SK,
    I.I_CATEGORY,
    SS.SS_QUANTITY,
    SS.SS_SALES_PRICE,
    SS.SS_NET_PROFIT
FROM SNOWFLAKE_SAMPLE_DATA.TPCDS_SF10TCL.STORE_SALES SS
JOIN SNOWFLAKE_SAMPLE_DATA.TPCDS_SF10TCL.CUSTOMER C
ON SS.SS_CUSTOMER_SK = C.C_CUSTOMER_SK

JOIN SNOWFLAKE_SAMPLE_DATA.TPCDS_SF10TCL.DATE_DIM D
ON SS.SS_SOLD_DATE_SK = D.D_DATE_SK

JOIN SNOWFLAKE_SAMPLE_DATA.TPCDS_SF10TCL.ITEM I
ON SS.SS_ITEM_SK = I.I_ITEM_SK
LIMIT 10000;


--Marketing Role Permissions Read Only
GRANT USAGE
ON DATABASE RETAIL_MARKETING_DB
TO ROLE MARKETING_ANALYST;

GRANT USAGE
ON SCHEMA RETAIL_MARKETING_DB.PUBLIC
TO ROLE MARKETING_ANALYST;

GRANT SELECT
ON TABLE RETAIL_MARKETING_DB.PUBLIC.CAMPAIGN_SALES
TO ROLE MARKETING_ANALYST;

--Finance Role Permissions Read Only
GRANT USAGE
ON DATABASE RETAIL_MARKETING_DB
TO ROLE FINANCE_USER;

GRANT USAGE
ON SCHEMA RETAIL_MARKETING_DB.PUBLIC
TO ROLE FINANCE_USER;

GRANT SELECT
ON TABLE RETAIL_MARKETING_DB.PUBLIC.CAMPAIGN_SALES
TO ROLE FINANCE_USER;


--ETL Developer Permissions Read + Modify
GRANT USAGE
ON DATABASE RETAIL_MARKETING_DB
TO ROLE ETL_DEVELOPER;

GRANT USAGE
ON SCHEMA RETAIL_MARKETING_DB.PUBLIC
TO ROLE ETL_DEVELOPER;

GRANT ALL PRIVILEGES
ON TABLE RETAIL_MARKETING_DB.PUBLIC.CAMPAIGN_SALES
TO ROLE ETL_DEVELOPER;

GRANT CREATE TABLE
ON SCHEMA RETAIL_MARKETING_DB.PUBLIC
TO ROLE ETL_DEVELOPER;

--Verify Permissions
SHOW GRANTS TO ROLE MARKETING_ANALYST;

SHOW GRANTS TO ROLE FINANCE_USER;

SHOW GRANTS TO ROLE ETL_DEVELOPER;

--Set Warehouse to XSMALL
ALTER WAREHOUSE FINANCE_WH
SET WAREHOUSE_SIZE = 'XSMALL';


USE WAREHOUSE FINANCE_WH;

--Run Sales Analysis Query
SELECT
    I_CATEGORY,
    SUM(SS_SALES_PRICE) AS TOTAL_SALES,
    SUM(SS_NET_PROFIT) AS TOTAL_PROFIT
FROM CAMPAIGN_SALES
GROUP BY I_CATEGORY
ORDER BY TOTAL_SALES DESC;


--Scale Warehouse to SMALL
ALTER WAREHOUSE FINANCE_WH
SET WAREHOUSE_SIZE = 'SMALL';

SELECT
    I_CATEGORY,
    SUM(SS_SALES_PRICE) AS TOTAL_SALES,
    SUM(SS_NET_PROFIT) AS TOTAL_PROFIT
FROM CAMPAIGN_SALES
GROUP BY I_CATEGORY
ORDER BY TOTAL_SALES DESC;

SHOW WAREHOUSES LIKE 'FINANCE_WH';

--Scale down again
ALTER WAREHOUSE FINANCE_WH
SET WAREHOUSE_SIZE = 'XSMALL';


--Run SAME Query Again
SELECT
    I_CATEGORY,
    AVG(SS_SALES_PRICE) AS AVG_SALES
FROM CAMPAIGN_SALES
GROUP BY I_CATEGORY;


SELECT MIN(SALE_DATE), MAX(SALE_DATE)
FROM CAMPAIGN_SALES;

SELECT DISTINCT SALE_DATE
FROM CAMPAIGN_SALES
ORDER BY SALE_DATE
LIMIT 20;

SELECT
    SALE_DATE,
    SUM(SS_SALES_PRICE) AS TOTAL_SALES
FROM CAMPAIGN_SALES
WHERE SALE_DATE BETWEEN '1998-01-01' AND '1998-12-31'
GROUP BY SALE_DATE
ORDER BY SALE_DATE;

--Add Cluster Key
ALTER TABLE CAMPAIGN_SALES
CLUSTER BY (SALE_DATE);
